"""M07 WO-8b: the `t08-rc-v1` records of the 0.2.0 RC job's local steps 11-17 (M07 spec §8; steps
11 and 12 are the alpha's, V02 alpha release spec §8.2, M07a.A40, A41, A43, judged exactly as the
alpha's `m07a_rc_external.py` judged them, check ids unchanged; steps 13-17 are M07.A60-A62).

Steps 11 and 12 run M01's and M02's own scripts, whose records carry no `commit` or `tree_clean`.
This script, run from outside the tree against the RC worktree at `C`, reads those records and
the exit codes of the commands that wrote them (`run_local.sh`'s summary), judges them by the
rules the M01 and M02 manifests already use (M01: `scripts/m01_evidence_manifest.py`'s `_idaes`
comparison; M02: `scripts/m02_evidence_manifest.py`'s `g10`, `g11`, `g12`, imported, not
restated), and writes one `t08-rc-v1` record per step through `t08_rc.finish`, so each carries
`commit`, `tree_clean`, the lock's SHA-256 and the machine class.

Steps 13-17 run their packages' own commands (`run_local.sh`); their records judge those
commands' exit codes, junit totals and registered output lines. Step 17's record is written into
its `$S` (§22.2: nothing under the repository is written).

    <rc venv>/bin/python m07_rc_external.py <step> --rc <rc worktree> --commit C --summary <summary.txt>
"""

from __future__ import annotations

def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def exit_codes(summary: Path) -> dict[str, int]:
    """The last exit code of each label in `run_local.sh`'s summary."""
    codes: dict[str, int] = {}
    for line in summary.read_text(encoding="utf-8").splitlines():
        match = re.match(r"^(\S+) exit (\d+) :: ", line)
        if match:
            codes[match[1]] = int(match[2])
    return codes


def differences(a: Any, b: Any, path: str = "") -> list[str]:
    """`m01_evidence_manifest._differences`, as a list."""
    import m01_evidence_manifest  # noqa: PLC0415

    return sorted(m01_evidence_manifest._differences(a, b, path))


def step11(rc: Path, out_dir: Path, codes: dict[str, int], t08_rc: Any) -> int:
    checks: list[dict[str, Any]] = []
    stdout = (out_dir / "external-crosscheck-check.stdout.txt").read_text(encoding="utf-8")
    t08_rc.check(
        checks,
        "a40.crosscheck_exit",
        codes.get("step11-xcheck") == 0,
        exit_code=codes.get("step11-xcheck"),
        last_line=stdout.strip().splitlines()[-1] if stdout.strip() else "",
        record_sha256=sha256(rc / "benchmarks/m01/external-crosscheck.json"),
        freeze_sha256=sha256(out_dir / "venv-xcheck.freeze.txt"),
    )
    committed_path = rc / "benchmarks/m01/idaes-conformance.json"
    rerun_path = out_dir / "idaes-conformance.json"
    ran = codes.get("step11-idaes") == 0 and rerun_path.exists()
    t08_rc.check(checks, "a40.idaes_exit", ran, exit_code=codes.get("step11-idaes"))
    differing = (
        differences(
            json.loads(committed_path.read_text(encoding="utf-8")),
            json.loads(rerun_path.read_text(encoding="utf-8")),
        )
        if ran
        else None
    )
    inputs_only = differing is not None and all(
        re.fullmatch(r"inputs\.[A-Za-z0-9_]+_sha256", member) for member in differing
    )
    t08_rc.check(
        checks,
        "a40.idaes_numbers_identical",
        inputs_only,
        differing_members=differing,
        committed_sha256=sha256(committed_path),
        rerun_sha256=sha256(rerun_path) if rerun_path.exists() else None,
        freeze_sha256=sha256(out_dir / "venv-idaes.freeze.txt"),
    )
    return int(
        t08_rc.finish(
            out_dir / "rc-step11.json",
            "external-records",
            checks,
            provenance={**t08_rc.provenance(), "machine_class": t08_rc.machine_class()},
        )
    )


def _junit(path: Path) -> dict[str, int]:
    totals = {"tests": 0, "failures": 0, "errors": 0, "skipped": 0}
    for suite in ElementTree.parse(path).getroot().iter("testsuite"):
        for key in totals:
            totals[key] += int(suite.get(key, "0"))
    return totals

def _log(logs: Path, label: str) -> str:
    path = logs / f"{label}.log"
    return path.read_text(encoding="utf-8", errors="replace") if path.exists() else ""


def _last(text: str) -> str:
    lines = [line for line in text.strip().splitlines() if line.strip()]
    return lines[-1] if lines else ""


def _tests(checks: list[dict[str, Any]], t08_rc: Any, check_id: str, codes: dict[str, int],
           label: str, junit: Path, command: str) -> None:
    totals = _junit(junit) if junit.exists() else None
    t08_rc.check(
        checks,
        check_id,
        codes.get(label) == 0 and totals is not None and totals["failures"] == totals["errors"] == 0,
        exit_code=codes.get(label),
        junit=totals,
        junit_sha256=sha256(junit) if junit.exists() else None,
        command=command,
    )


def _exit(checks: list[dict[str, Any]], t08_rc: Any, check_id: str, codes: dict[str, int],
          label: str, logs: Path, **value: Any) -> None:
    t08_rc.check(
        checks, check_id, codes.get(label) == 0, exit_code=codes.get(label),
        last_line=_last(_log(logs, label)), **value
    )


def _finish(t08_rc: Any, out: Path, step: str, checks: list[dict[str, Any]]) -> int:
    return int(
        t08_rc.finish(
            out, step, checks,
            provenance={**t08_rc.provenance(), "machine_class": t08_rc.machine_class()},
        )
    )

NODE_LINE = re.compile(r"^tests/\S+::")


def nlp_selection(selected: str, everything: str, selected_exit: int | None,
                  modules: list[str]) -> tuple[bool, dict[str, Any]]:
    """V03.A33 (R-907): `-m nlp` collects the same node ids over the `mark.nlp` modules as over
    `tests/`, 190 of them, and the run over the modules exits 0 (the run over `tests/` exits 2 on
    the `mpmath` modules; its status is not judged)."""
    ids_selected = {line.strip() for line in selected.splitlines() if NODE_LINE.match(line)}
    ids_all = {line.strip() for line in everything.splitlines() if NODE_LINE.match(line)}
    equal = ids_selected == ids_all
    value = {
        "modules": modules,
        "selected_count": len(ids_selected),
        "all_count": len(ids_all),
        "equal": equal,
        "selected_exit_code": selected_exit,
    }
    return (equal and len(ids_selected) == 190 and selected_exit == 0 and bool(modules)), value

#: V03.A32 (c): the 25 registered testcases (junit classname, name).
G15_NODES: tuple[tuple[str, str], ...] = (

    ("tests.test_m06_w27_coverage", "test_committed_coverage_json_is_g14_and_this_builds_classification"),
)
#: V03.A32 (d): the dry run's registered refusal (W27-R63), its last non-empty line exactly.
W27_V1_REFUSAL = (
    "benchmarks.m06.w27.snapshot.SnapshotUnsupportedError: registry_snapshot: unsupported(provider "
    "'iapws95-v1': no registered records and phases, W27-R63)"
)


def g15_registered_states(junit: Path) -> tuple[bool, dict[str, Any]]:
    """V03.A32 (c): (a)'s junit has 0 skipped, and each of the 25 testcases appears exactly once
    with no `failure`, `error` or `skipped` child."""
    if not junit.exists():
        return False, {"nodes": {}, "skipped_total": None, "reason": "no junit"}
    cases = list(ElementTree.parse(junit).getroot().iter("testcase"))
    skipped_total = sum(1 for case in cases if case.find("skipped") is not None)
    nodes: dict[str, str] = {}
    for classname, name in G15_NODES:
        found = [c for c in cases if c.get("classname") == classname and c.get("name") == name]
        key = f"{classname}::{name}"
        if len(found) != 1:
            nodes[key] = "missing" if not found else f"{len(found)} times"
            continue
        tags = {child.tag for child in found[0]}
        nodes[key] = next((t for t in ("failure", "error", "skipped") if t in tags), "passed")
        if nodes[key] == "failure":
            nodes[key] = "failed"
    ok = skipped_total == 0 and len(nodes) == 25 and set(nodes.values()) == {"passed"}
    return ok, {"nodes": nodes, "skipped_total": skipped_total}


def w27_v1_refusal(exit_code: int | None, log: str, summary_written: bool) -> tuple[bool, dict[str, Any]]:
    """V03.A32 (d): the W27 v1 dry run exits 1, its last non-empty line is the registered
    W27-R63 refusal exactly, and it writes no `summary.json`."""
    last = _last(log)
    value = {"exit_code": exit_code, "last_line": last, "summary_written": summary_written}
    return (exit_code == 1 and last == W27_V1_REFUSAL and not summary_written), value

LATER = {"step13": step13, "step14": step14, "step15": step15, "step16": step16, "step17": step17}


STEPS = ("step11", "step12", "step13", "step14", "step15", "step16", "step17")


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("step", choices=STEPS)
    parser.add_argument("--rc", type=Path, required=True)
    parser.add_argument("--commit", required=True)
    parser.add_argument("--summary", type=Path, required=True)
    parser.add_argument("--logs", type=Path, help="run_local.sh's log directory (steps 13-17)")
    parser.add_argument("--s17", type=Path, help="step 17's $S")
    arguments = parser.parse_args()
    rc = arguments.rc.resolve()
    sys.path.insert(0, str(rc / "scripts"))
    import t08_rc  # noqa: PLC0415

    if t08_rc.git("rev-parse", "HEAD").strip() != arguments.commit:
        print(f"refused: {rc} is not at {arguments.commit}", file=sys.stderr)
        return 2
    out_dir = rc / "evidence/S13" / arguments.commit / "artifacts/rc" / arguments.step
    codes = exit_codes(arguments.summary)
    if arguments.step in ("step11", "step12"):
        run = step11 if arguments.step == "step11" else step12
        return run(rc, out_dir, codes, t08_rc)
    if arguments.step == "step17":
        out_dir = arguments.s17
    out_dir.mkdir(parents=True, exist_ok=True)
    return LATER[arguments.step](rc, out_dir, codes, t08_rc, arguments.logs, arguments.commit)
