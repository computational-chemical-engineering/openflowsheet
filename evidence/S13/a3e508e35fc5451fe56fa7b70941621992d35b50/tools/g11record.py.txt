"""V03 WO-7b (V03 spec §4 "Step 12's G11 results", V03.A34, R-909): the record of G11's keep-and-audit at C.

    python g11record.py <artifacts/rc/step12 dir> <run_g11keep.out> <g11equal.py>     (cwd: the RC worktree at C)

Writes `<step12>/experiments/rc-step12-experiments.json` (format `t08-rc-v1`, step `g11-keep-and-audit`, the provenance
members of `rc-step12.json`) with exactly V03.A34's three checks:
(a) `r375.g11_rerun`: the re-run (`run_g11keep.sh`) exited 0;
(b) `r375.same_computation`: g11equal.py's check (1) — the re-run's runs file and step 12's own (SHA-256 `cdb7d999…`)
    have 2 runs each, the same labels in the same order, and 0 differences in `stage`, `accepted`, `child_accepted`,
    `tube_outlet`, `path`, `attempts` over the 242 point-runs;
(c) `r375.audit`: `scripts/m07_g11_experiments_audit.py --runs <step 12's own runs file>` over the two collected
    records exits 0 and prints exactly A34's line, with the job_id count j free (recorded, not judged).
"""

from __future__ import annotations

import hashlib
import re
import subprocess
import sys
from pathlib import Path

STEP12_RUNS_SHA256 = "cdb7d9993327cd48543212c37063421fa9c82232dd05b356e1609335fce6fd0a"
LINE = re.compile(
    r"^G11 experiments audit: 2 runs, 210 boundary points, 210 results, 210 attempts, 0 without a result, "
    r"32 direct points \(no records\); 6 items \+ schema: 0 problems; context \(job_id\): (\d+) findings: PASS$"
)


def main() -> int:
    step12, out_file, equal = (Path(p).resolve() for p in sys.argv[1:4])
    sys.path.insert(0, str(Path("scripts").resolve()))
    import t08_rc  # noqa: PLC0415

    exp = step12 / "experiments"
    own = step12 / "g11-coverage-v3.runs.json"
    rerun = exp / "g11-coverage-v3.runs.json"
    records = [exp / "g11-experiments-run1.json", exp / "g11-experiments-run2.json"]
    checks: list[dict] = []

    found = re.findall(r"^EXIT step12-g11-keep exit (\d+) (\d+)s$", out_file.read_text("utf-8"), re.MULTILINE)
    code = int(found[-1][0]) if len(found) == 1 else None
    t08_rc.check(checks, "r375.g11_rerun", code == 0, exit_code=code,
                 wall_s=int(found[-1][1]) if len(found) == 1 else None,
                 command="tools/run_g11keep.sh (step12-g11's command through tools/g11keep.py.txt)")

    own_sha = hashlib.sha256(own.read_bytes()).hexdigest()
    eq = subprocess.run([sys.executable, str(equal), str(own), str(rerun), *map(str, records)],
                        capture_output=True, text=True, check=False)
    lines = eq.stdout.splitlines()
    first = next((line for line in lines if line.startswith("(1) ")), "")
    m = re.match(r"^\(1\) point-runs compared (\d+), fields .*, differences (\d+)$", first)
    t08_rc.check(checks, "r375.same_computation",
                 own_sha == STEP12_RUNS_SHA256 and m is not None and m.group(1) == "242" and m.group(2) == "0",
                 step12_runs_sha256=own_sha, rerun_runs_sha256=hashlib.sha256(rerun.read_bytes()).hexdigest(),
                 point_runs=int(m.group(1)) if m else None, differences=int(m.group(2)) if m else None,
                 members=["stage", "accepted", "child_accepted", "tube_outlet", "path", "attempts"],
                 equal_exit_code=eq.returncode, equal_output=lines)

    audit_out = exp / "g11-experiments-audit.json"
    command = [sys.executable, "scripts/m07_g11_experiments_audit.py", "--variant-file",
               "benchmarks/m02/variant-v3-provisional.json", "--runs", str(own), "--out", str(audit_out),
               *map(str, records)]
    au = subprocess.run(command, capture_output=True, text=True, check=False)
    last = au.stdout.splitlines()[-1] if au.stdout.strip() else ""
    lm = LINE.match(last)
    t08_rc.check(checks, "r375.audit", au.returncode == 0 and lm is not None, exit_code=au.returncode,
                 last_line=last, job_id_findings=int(lm.group(1)) if lm else None,
                 runs_file="step12/g11-coverage-v3.runs.json",
                 command="python scripts/m07_g11_experiments_audit.py --variant-file "
                 "benchmarks/m02/variant-v3-provisional.json --runs <rc>/step12/g11-coverage-v3.runs.json "
                 "--out <rc>/step12/experiments/g11-experiments-audit.json "
                 "<rc>/step12/experiments/g11-experiments-run1.json <rc>/step12/experiments/g11-experiments-run2.json")
    return int(t08_rc.finish(exp / "rc-step12-experiments.json", "g11-keep-and-audit", checks,
                             provenance={**t08_rc.provenance(), "machine_class": t08_rc.machine_class()}))


if __name__ == "__main__":
    sys.exit(main())
