def verdict(path: Path) -> str:
    record = json.loads(path.read_text(encoding="utf-8"))
    checks = record["checks"]
    passed = sum(check["result"] == "pass" for check in checks)
    word = "PASS" if record.get("passed") is True and passed == len(checks) else "FAIL"
    return f"{word} ({passed}/{len(checks)})"


def main() -> int:
    if sys.argv[1] == "cited":
        paths = [path for row in ROWS for path in row[4]]
        Path(sys.argv[2]).write_text("\n".join(paths) + "\n", encoding="utf-8")
        print(f"{len(paths)} paths, {len(set(paths))} distinct")
        return 0 if len(paths) == len(set(paths)) else 2
    rc, artifacts = Path(sys.argv[2]), Path(sys.argv[3])
    index = {f["path"]: f for f in json.loads((rc / "records.json").read_text("utf-8"))["files"]}
    cited = [path for row in ROWS for path in row[4]]
    if sorted(cited) != sorted(index):
        print(f"cited and records.json differ: {sorted(set(cited) ^ set(index))}", file=sys.stderr)
        return 2
    print("| Step | Assertion | Where | Result (as the record says) | Record | sha256 |")
    print("| --- | --- | --- | --- | --- | --- |")
    for step, assertion, where, note, paths, record in ROWS:
        result = note if record is None else f"{verdict(artifacts / record)}: {note}"
        shown = ", ".join(f"`{p}`" for p in paths) or "`ci/job-logs/check-ubuntu-latest.log` (step 1's record)"
        hashes = ", ".join(f"`{index[p]['sha256']}`" for p in paths) or "— (cited once, on step 1's row)"
        print(f"| {step} | {assertion} | {where} | {result} | {shown} | {hashes} |")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
