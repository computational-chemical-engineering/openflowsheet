def row(step: int, assertion: str, where: str, result: str, paths: list[str]) -> str:
    if not paths:
        record, digest = "`ci/job-logs/check-ubuntu-latest.log`", f"`{INDEX['ci/job-logs/check-ubuntu-latest.log']['sha256']}`"
    else:
        record = ", ".join(f"`{p}`" for p in paths)
        digest = ", ".join(f"`{INDEX[p]['sha256']}`" for p in paths)
    return f"| {step} | {assertion} | {where} | {result} | {record} | {digest} |"


cited = [p for *_, paths in ROWS for p in paths]
assert len(cited) == len(set(cited)), "a record is cited twice"
assert set(cited) == set(INDEX), sorted(set(INDEX) ^ set(cited))
table = "\n".join(row(*r) for r in ROWS)
template = (Path(__file__).with_name("rc-record.template.md")).read_text("utf-8")
out = template.replace("@TABLE@", table).replace("@N@", str(len(INDEX)))
(ROOT / "docs" / "v02-rc-record.md").write_text(out, encoding="utf-8")
print(f"{len(ROWS)} rows, {len(INDEX)} records cited")
